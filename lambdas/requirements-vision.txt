# The vision worker's own dependencies (vision/ and vision_worker_handler.py). Kept out of
# requirements.txt for the same reason as requirements-ops-mcp.txt: that file is the one zip every
# pipeline Lambda shares, and none of them need OpenCV. The worker is packaged on its own, for arm64
# (docs/enhancements/opencv-agentic-vision-enhancement.md §4).
#
# opencv-python-headless 5.0.0.93: stock OpenCV 5, with manylinux2014 aarch64 and x86_64 wheels.
# This is the baseline backend; the COOL backend swaps this cv2 for OpenCV's Graviton build and
# runs the same code.
#
# numpy 2.4.4: the newest release that still supports Python 3.11 (CI's version). Its wheels need
# glibc 2.27+, which is why the worker runs on Lambda's python3.12 runtime (Amazon Linux 2023), not
# python3.11 (Amazon Linux 2, glibc 2.26).
#
# requests: the worker's HTTP range reads (vision_worker_handler.http_fetch). The same pin as
# requirements.txt; the worker's package installs this file alone.
#
# networkx 3.4.2: the rail access task's graph measures (vision/network.py and rail_analyse.py:
# betweenness, shortest paths to hubs, simulated links). Pure Python, one py3-none-any wheel for
# 3.11, 3.12 and 3.13 alike, and in the worker only: the adapters never import it.
opencv-python-headless==5.0.0.93
numpy==2.4.4
requests==2.34.2
networkx==3.4.2
